# Step 1: Create a new project folder (if not already created)
mkdir Flask-Backend-Project
cd Flask-Backend-Project

# Step 2: Create a Python virtual environment
python -m venv env

# Step 3: Activate the virtual environment
# Windows:
env\Scripts\activate
# macOS/Linux:
# source env/bin/activate

# Step 4: Install Flask and other dependencies
pip install flask

# Optional: freeze installed packages to requirements.txt
pip freeze > requirements.txt

# Step 5: Create the main Flask application file
# Example: app.py
# File content:
# from flask import Flask
# app = Flask(__name__)
# @app.route('/')
# def home():
#     return "Hello, Flask!"
# if __name__ == '__main__':
#     app.run(debug=True)

# Step 6: Run the Flask development server
# Default runs at http://127.0.0.1:5000/
flask run
# Or, if using app.py:
python app.py

# Step 7: (Optional) Install Flask extensions for database, CORS, etc.
pip install flask_sqlalchemy flask_cors flask_migrate

# Step 8: Organize project structure (folders)
# Recommended structure:
# Flask-Backend-Project/
# ├── app.py
# ├── requirements.txt
# ├── env/
# ├── app/
# │   ├── __init__.py
# │   ├── routes/
# │   ├── models/
# │   ├── services/
# │   └── utils/
# └── migrations/ (if using Flask-Migrate)
